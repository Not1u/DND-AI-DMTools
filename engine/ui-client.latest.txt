const __rawCall = (typeof host !== 'undefined' && host && typeof host.call === 'function') ? host.call.bind(host) : null
const __bridgeCall = async (name, args) => {
  try {
    if (typeof fetch !== 'function') return null
    const res = await fetch('/dnd5e/api', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ op: name, args: args === undefined ? null : args }),
    })
    if (!res.ok) return null
    const data = await res.json()
    if (!data || data.ok !== true) return null
    return data.value
  } catch (e) { return null }
}
const call = async (name, args) => {
  const viaBridge = await __bridgeCall(name, args)
  if (viaBridge !== null) return viaBridge
  if (__rawCall) return __rawCall(name, args)
  throw new Error('[dnd-5ednd] no transport for ' + name)
}

function LevelSet(props) {
  const s = props.s
  const manual = !!props.manual
  const cur = Number(s.level) || 1
  const autoTarget = Math.min(20, cur + 1)
  const xpNext = (s.xpNext == null ? null : Number(s.xpNext))
  const canLevel = xpNext == null ? true : (Number(s.xp) >= xpNext)
  const xpShort = xpNext == null ? 0 : Math.max(0, xpNext - Number(s.xp || 0))
  const [target, setTarget] = React.useState(manual ? cur : autoTarget)
  const [syncXp, setSyncXp] = React.useState(true)
  const [open, setOpen] = React.useState(!manual)
  const [info, setInfo] = React.useState(null)
  const [states, setStates] = React.useState([])
  const [busy, setBusy] = React.useState(false)
  const [msg, setMsg] = React.useState('')
  const effTarget = manual ? target : autoTarget
  const calc = React.useCallback(() => {
    setMsg('')
    return call('levelset.info', { id: s.id, target: effTarget }).then(r => { if (r && r.ok) { setInfo(r); setStates((r.pending || []).map(() => ({}))) } else setMsg((r && r.error) || '计算失败') })
  }, [s.id, effTarget])
  // 自动升级模式：进面板就算好要选什么
  React.useEffect(() => { if (!manual && canLevel) calc() }, [s.id, s.level, s.xp, canLevel])
  const doApply = async () => {
    setBusy(true); setMsg('')
    try {
      const choices = (info && info.pending || []).map((ch, i) => { const st = states[i] || {}; if (ch.kind === 'subclass') return { kind: 'subclass', key: st.key, picks: st.picks }; if (ch.kind === 'pactBoon') return { kind: 'pactBoon', key: st.key }; if (ch.kind === 'invocation') return { kind: 'invocation', keys: st.keys }; if (ch.kind === 'asi') return Object.assign({ kind: 'asi' }, st); if (ch.kind === 'cantrip' || ch.kind === 'spell') return { kind: ch.kind, keys: st.keys }; return null }).filter(x => x && (x.key || x.keys || x.mode))
      const payload = { id: s.id, target: effTarget, choices: choices }
      if (manual && syncXp) payload.setXp = true
      const r = await call('levelset.apply', { json: JSON.stringify(payload) })
      if (r && r.ok) { setMsg(r.summary + (r.saved ? '（已保存）' : '（未保存：' + (r.saveError || '未知原因') + '）')); setInfo(null); if (props.onReload) props.onReload() } else setMsg((r && r.error) || '应用失败')
    } catch (e) { setMsg(String(e && e.message || e)) }
    setBusy(false)
  }
  const kids = []
  if (manual) {
    kids.push(h('div', { key: 'h', style: { display: 'flex', alignItems: 'center', gap: 8, flexWrap: 'wrap' } },
      h('b', null, '手动调整等级'),
      h('span', { className: 'dndp-chip' }, 'DM 覆写'),
      h('div', { className: 'dndp-sp' }),
      h('button', { className: 'dndp-btn tiny', onClick: () => setOpen(!open) }, open ? '收起' : '展开')))
    if (!open) return h('div', { className: 'dndp-panel' }, kids)
    kids.push(h('div', { key: 'w', className: 'dndp-warn', style: { marginTop: 6 } },
      '这里直接设定等级，不走经验。跨过需要做选择的等级（例如 Lv3 的子职、Lv4 的属性提升）时请改用上面的「升级（按经验）」。'))
    kids.push(h('div', { key: 'p', style: { marginTop: 6, display: 'flex', alignItems: 'center', gap: 8, flexWrap: 'wrap' } },
      h('button', { className: 'dndp-ico', onClick: () => setTarget(t => Math.max(1, t - 1)) }, '−'),
      h('b', null, 'Lv' + target),
      h('button', { className: 'dndp-ico', onClick: () => setTarget(t => Math.min(20, t + 1)) }, '+'),
      h('input', { className: 'dndp-input', style: { width: 62 }, value: String(target), onChange: e => setTarget(Math.max(1, Math.min(20, Number(e.target.value) || 1))) }),
      h('label', { style: { display: 'inline-flex', alignItems: 'center', gap: 5, fontSize: 12 } },
        h('input', { type: 'checkbox', checked: syncXp, onChange: e => setSyncXp(e.target.checked) }), '同时把经验设为该等级门槛'),
      h('button', { className: 'dndp-btn', onClick: calc }, '计算影响')))
  } else {
    kids.push(h('div', { key: 'h', style: { display: 'flex', alignItems: 'center', gap: 8, flexWrap: 'wrap' } },
      h('b', null, '升级（按经验）'),
      h('span', { className: 'dndp-chip' }, 'Lv' + cur + ' → Lv' + autoTarget),
      h('div', { className: 'dndp-sp' }),
      h('button', { className: 'dndp-btn tiny', onClick: calc }, '重算')))
    if (!canLevel) {
      kids.push(h('div', { key: 'x', className: 'dndp-note', style: { marginTop: 6 } },
        '经验还没满：' + (s.xp || 0) + ' / ' + (xpNext == null ? '—' : xpNext) + '，还差 ' + xpShort + ' 点；升不了级。（DM 可以用下面的「手动调整等级」直接覆写。）'))
      if (msg) kids.push(h('div', { key: 'm', className: 'dndp-note hl', style: { marginTop: 6 } }, msg))
      return h('div', { className: 'dndp-panel' }, kids)
    }
  }
  if (info && info.ok) {
    if (info.same) kids.push(h('div', { key: 's', className: 'dndp-note', style: { marginTop: 6 } }, '当前已是 Lv' + info.level))
    else {
      kids.push(h('div', { key: 'd', className: 'dndp-note', style: { marginTop: 6 } }, 'Lv' + info.level + ' → Lv' + info.target + '：每级 HP ' + (info.hpPerLevel >= 0 ? '+' : '') + info.hpPerLevel))
      if (info.slotsAfter && info.slotsAfter.length) kids.push(h('div', { key: 'sl', className: 'dndp-note' }, '调整后法术位：' + info.slotsAfter.map(x => x.lv + '环×' + x.max).join('、')))
      ;(info.pending || []).forEach((ch, i) => kids.push(h('div', { key: 'p' + i, style: { marginTop: 8 } }, h(ChoiceBox, { ch: ch, st: states[i] || {}, setSt: v => setStates(arr => { const n = arr.slice(); n[i] = v; return n }) }))))
      if (!info.pending || !info.pending.length) kids.push(h('div', { key: 'n', className: 'dndp-note' }, '该等级区间无需额外选择'))
      kids.push(h('div', { key: 'go', style: { marginTop: 10 } },
        h('button', { className: 'dndp-btn' + (manual ? '' : ' primary'), disabled: busy, onClick: doApply }, manual ? ('覆写为 Lv' + effTarget + (syncXp ? '（并同步经验）' : '')) : ('升级到 Lv' + effTarget))))
    }
  }
  if (msg) kids.push(h('div', { key: 'm', className: 'dndp-note hl', style: { marginTop: 6 } }, msg))
  return h('div', { className: 'dndp-panel' }, kids)
}

function RulesPanel(props) {
  const [q, setQ] = React.useState('')
  const [book, setBook] = React.useState('')
  const [items, setItems] = React.useState(null)
  const [stats, setStats] = React.useState(null)
  const [openId, setOpenId] = React.useState('')
  const [text, setText] = React.useState('')
  const [busy, setBusy] = React.useState(false)
  const [msg, setMsg] = React.useState('')
  const [recent, setRecent] = React.useState(() => { try { const v = localStorage.getItem('dnd5e.recentRules'); return v ? JSON.parse(v) : [] } catch (e) { return [] } })
  React.useEffect(() => {
    call('rules.stats', {}).then(r => { if (r && r.ok) setStats(r); else setMsg('规则索引不可用：' + ((r && r.error) || '') + '（在仓库运行 node tools/index-rules.mjs 生成索引）') }).catch(e => setMsg(String(e && e.message || e)))
  }, [])
  const remember = (term) => {
    const next = [term].concat(recent.filter(x => x !== term)).slice(0, 6)
    setRecent(next)
    try { localStorage.setItem('dnd5e.recentRules', JSON.stringify(next)) } catch (e) { }
  }
  const terms = () => q.trim().split(/[\s,，]+/).filter(Boolean)
  const run = async (override) => {
    const query = typeof override === 'string' ? override : q
    if (!query.trim()) return
    if (override) setQ(query)
    setBusy(true); setMsg(''); setItems(null); setOpenId(''); setText('')
    try {
      const args = { query: query, limit: 12 }
      if (book) args.book = book
      const r = await call('rules.search', args)
      if (r && r.ok) { setItems(r.items || []); remember(query.trim()); if (!(r.items || []).length) setMsg('未命中：' + query) } else setMsg((r && r.error) || '检索失败')
    } catch (e) { setMsg(String(e && e.message || e)) }
    setBusy(false)
  }
  const open = async (id) => {
    if (openId === id) { setOpenId(''); return }
    setOpenId(id); setText('读取中…')
    try {
      const r = await call('rules.read', { id: id })
      setText(r && r.ok ? r.entry.text : ('读取失败：' + ((r && r.error) || '')))
    } catch (e) { setText(String(e && e.message || e)) }
  }
  const hl = (str) => {
    const ts = terms().filter(t => t.length > 0)
    if (!ts.length) return str
    const out = []
    let rest = String(str)
    let guard = 0
    while (rest.length && guard++ < 200) {
      const lower = rest.toLowerCase()
      let best = -1, bestLen = 0
      for (const t of ts) { const i = lower.indexOf(t.toLowerCase()); if (i >= 0 && (best < 0 || i < best)) { best = i; bestLen = t.length } }
      if (best < 0) { out.push(rest); break }
      if (best > 0) out.push(rest.slice(0, best))
      out.push(h('span', { key: 'm' + guard, className: 'dndp-hl' }, rest.slice(best, best + bestLen)))
      rest = rest.slice(best + bestLen)
    }
    return out
  }
  const kids = []
  kids.push(h('div', { key: 'h', style: { display: 'flex', alignItems: 'center', gap: 8 } },
    h('b', null, '规则查找'),
    h('div', { className: 'dndp-sp' }),
    stats ? h('span', { className: 'dndp-chip' }, (stats.total || 0) + ' 条 / ' + ((stats.books || []).length) + ' 本书') : null))
  kids.push(h('div', { key: 'f', style: { display: 'flex', gap: 6, marginTop: 8 } },
    h('input', { key: 'i', className: 'dndp-input', placeholder: '关键词，空格=同时满足，如：擒抱 体型 / 火球术 伤害', value: q, onChange: e => setQ(e.target.value), onKeyDown: e => { if (e.key === 'Enter') run() } }),
    h('button', { key: 'b', className: 'dndp-btn primary', disabled: busy, onClick: () => run() }, '检索')))
  if (recent.length) kids.push(h('div', { key: 'rec', className: 'dndp-tags', style: { marginTop: 6 } },
    h('span', { key: 'lb', className: 'dndp-note' }, '最近：'),
    recent.map((r2, i) => h('button', { key: i, className: 'dndp-tag', style: { cursor: 'pointer' }, onClick: () => run(r2) }, r2))))
  if (stats && (stats.books || []).length) {
    const chips = [h('button', { key: 'all', className: 'dndp-tag', style: { cursor: 'pointer', borderColor: book === '' ? 'var(--dsw-alias-brand-primary)' : undefined }, onClick: () => setBook('') }, '全部')]
    ;(stats.books || []).forEach((b, i) => chips.push(h('button', { key: i, className: 'dndp-tag', style: { cursor: 'pointer', borderColor: book === b.book ? 'var(--dsw-alias-brand-primary)' : undefined }, onClick: () => setBook(book === b.book ? '' : b.book) }, b.book + ' ' + b.entries)))
    kids.push(h('div', { key: 'bk', className: 'dndp-tags', style: { marginTop: 6 } }, chips))
  }
  if (items && items.length) {
    kids.push(h('div', { key: 'l', className: 'dndp-label' }, '结果 ' + items.length + ' 条（点击展开全文）'))
    items.forEach((it, i) => {
      kids.push(h('div', { key: 'r' + i, className: 'dndp-sect', style: { marginTop: 6 } },
        h('button', { className: 'dndp-sec', onClick: () => open(it.id) },
          h('span', { className: 'dndp-chev' + (openId === it.id ? ' open' : '') }, '▶'),
          h('span', null, hl(it.title)),
          h('div', { className: 'dndp-sp' }),
          h('span', { className: 'dndp-chip' }, it.book)),
        openId === it.id
          ? h('div', { className: 'dndp-body' }, h('div', { style: { whiteSpace: 'pre-wrap', fontSize: 12, maxHeight: 340, overflow: 'auto' } }, text))
          : h('div', { className: 'dndp-body' }, h('div', { className: 'dndp-note' }, hl(it.snippet)))))
    })
  }
  if (msg) kids.push(h('div', { key: 'm', className: 'dndp-note hl' }, msg))
  return h('div', { className: 'dndp-panel' }, kids)
}

const MCLS_ABILITY = { barbarian: 'str', bard: 'cha', cleric: 'wis', druid: 'wis', fighter: 'str', monk: 'dex', paladin: 'str', ranger: 'dex', rogue: 'dex', sorcerer: 'cha', warlock: 'cha', wizard: 'int' }
// 5e 兼职前置：新职业主属性 ≥ 13，且所有已有职业主属性也 ≥ 13（属性值，不是修正）
function mcPrereqInfo(s, key, owned) {
  const own = (owned && owned.length) ? owned : (s.clsKeys || [])
  const need = []
  const newAb = MCLS_ABILITY[key]
  if (newAb) need.push({ ab: newAb, why: (CLSN[key] || key) + ' 的主属性' })
  for (const o of own) { const ab = MCLS_ABILITY[o]; if (ab && !need.some(x => x.ab === ab)) need.push({ ab: ab, why: (CLSN[o] || o) + '（已有职业）' }) }
  const bad = need.filter(x => ((s.mods || {})[x.ab] || 0) < 1)
  return { need: need, ok: bad.length === 0, bad: bad, text: need.map(x => ABF[x.ab] + '≥13').join('、') }
}

// 每格确定性微差：让重复瓦片不至于像贴瓷砖
function cellVariation(x, y) {
  let h = ((x + 1) * 73856093) ^ ((y + 1) * 19349663)
  h = (h >>> 0)
  const shade = (h % 7) * 0.012
  return {
    backgroundPosition: ((h % 23) - 11) + 'px ' + (((h >> 5) % 19) - 9) + 'px',
    boxShadow: 'inset 0 0 0 40px rgba(0,0,0,' + shade.toFixed(3) + ')',
  }
}
const cellName = (x, y) => {
  let s = '', n = Math.max(0, Math.floor(x))
  do { s = String.fromCharCode(65 + (n % 26)) + s; n = Math.floor(n / 26) - 1 } while (n >= 0)
  return s + (Math.floor(y) + 1)
}
function cellDist(a, b) {
  const dx = Math.abs(a.x - b.x), dy = Math.abs(a.y - b.y)
  const mx = Math.max(dx, dy), mn = Math.min(dx, dy)
  return { cells: mx, feetSimple: mx * 5, feetStandard: (mx + Math.floor(mn / 2)) * 5 }
}

const cloneLayout = (l) => JSON.parse(JSON.stringify(l || DEFAULT_LAYOUT))
function normLayout(l) {
  const out = cloneLayout(l && l.slots ? l : DEFAULT_LAYOUT)
  out.slots = out.slots || {}
  out.weights = out.weights || {}
  for (const [k] of WS_SLOTS) {
    if (!out.slots[k]) out.slots[k] = { size: k === 'left' ? 240 : (k === 'right' ? 380 : 0), panels: [] }
    if (!Array.isArray(out.slots[k].panels)) out.slots[k].panels = []
    out.slots[k].panels = out.slots[k].panels.filter(p => PANEL_DEFS[p])
    if (!Array.isArray(out.weights[k]) || out.weights[k].length !== out.slots[k].panels.length) out.weights[k] = out.slots[k].panels.map(() => 1)
  }
  out.collapsed = Array.isArray(out.collapsed) ? out.collapsed : []
  return out
}
const inLayout = (l, id) => WS_SLOTS.some(([k]) => l.slots[k].panels.indexOf(id) >= 0)
// 把面板移动到目标槽位的目标位置（同槽位即重排）。纯函数，方便单测。
function reorderPanels(base, pid, toSlot, toIndex) {
  const n = cloneLayout(base)
  if (!n.slots[toSlot] || !PANEL_DEFS[pid]) return n
  const fromSlot = slotOf(n, pid)
  if (fromSlot) {
    const fi = n.slots[fromSlot].panels.indexOf(pid)
    if (fi >= 0) { n.slots[fromSlot].panels.splice(fi, 1); n.weights[fromSlot].splice(fi, 1) }
    if (fromSlot === toSlot && fi >= 0 && fi < toIndex) toIndex--
  }
  const at = Math.max(0, Math.min(n.slots[toSlot].panels.length, toIndex))
  n.slots[toSlot].panels.splice(at, 0, pid)
  n.weights[toSlot].splice(at, 0, 1)
  n.weights[toSlot] = n.slots[toSlot].panels.map(() => 1)
  if (fromSlot && n.slots[fromSlot].panels.length) n.weights[fromSlot] = n.slots[fromSlot].panels.map(() => 1)
  n.collapsed = (n.collapsed || []).filter(x => x !== pid)
  return n
}
function slotOf(l, id) { for (const [k] of WS_SLOTS) if (l.slots[k].panels.indexOf(id) >= 0) return k; return null }

      if (y + need > window.innerHeight) y = Math.max(8, r.top - need - 4)
    }
    setMenu({ x: x, y: y })
  }
  const onTitleDown = (e) => {
    const t = e.target
    if (t && t.tagName === 'BUTTON') return
    if (e.button !== 0) return
    if (props.onDragStart) props.onDragStart(e)
  }
  const head = h('div', { className: 'dndp-ptitle' + (props.dragging ? ' dragging' : ''), onPointerDown: onTitleDown, title: '按住标题栏拖动可换位置' },
    h('span', null, def.icon), h('span', null, def.title), h('div', { className: 'sp' }),
    h('button', { className: 'dndp-pbtn', title: props.collapsed ? '展开' : '折叠', onPointerDown: e => e.stopPropagation(), onClick: () => props.onCollapse() }, props.collapsed ? '▸' : '▾'),
    h('button', { className: 'dndp-pbtn', title: props.isMax ? '还原' : '最大化', onPointerDown: e => e.stopPropagation(), onClick: () => props.onMax() }, props.isMax ? '❐' : '⤢'),
    h('button', { className: 'dndp-pbtn', title: '移动 / 关闭', onPointerDown: e => e.stopPropagation(), onClick: openMenu }, '⋯'))
  const items = [
    h('div', { key: 'h1', className: 'hd' }, '移到槽位'),
    ...WS_SLOTS.map(([k, n]) => h('button', { key: k, onClick: () => { props.onMove(k); setMenu(null) } }, n + (slotOf(props.layout, props.id) === k ? '（当前）' : ''))),
    h('div', { key: 's', className: 'sep' }),
    h('button', { key: 'add', onClick: () => { props.onAddTo(props.id); setMenu(null) } }, '复制到另一槽位'),
    h('button', { key: 'rm', onClick: () => { props.onClose(props.id); setMenu(null) } }, '从工作区移除'),
  ]
  return h('div', { className: 'dndp-panel' + (isCol ? ' col' : '') + (props.isMax ? ' max' : '') + (props.dragging ? ' dragsrc' : ''), style: props.style },
    head,
    menu ? h('div', { className: 'dndp-pmenu', style: { position: 'fixed', left: menu.x, top: menu.y, width: 196 }, onPointerDown: e => e.stopPropagation() }, items) : null,
    h('div', { className: 'dndp-pbody' }, isCol ? null : props.children))
}

function Workspace() {
  const [layout, setLayout] = React.useState(null)
  const [activeId, setActiveId] = React.useState(null)
  const [party, setParty] = React.useState([])
  const [toolCount, setToolCount] = React.useState(null)
  const [xpMap, setXpMap] = React.useState({})
  const [xpReady, setXpReady] = React.useState([])
  const [autoLevelOpened, setAutoLevelOpened] = React.useState(false)
  const [conditionCatalog, setConditionCatalog] = React.useState([
    { key: 'blinded', name: '目盲' }, { key: 'charmed', name: '魅惑' }, { key: 'deafened', name: '耳聋' },
    { key: 'exhaustion', name: '力竭' }, { key: 'frightened', name: '恐慌' }, { key: 'grappled', name: '擒抱' },
    { key: 'incapacitated', name: '失能' }, { key: 'invisible', name: '隐形' }, { key: 'paralyzed', name: '麻痹' },
    { key: 'petrified', name: '石化' }, { key: 'poisoned', name: '中毒' }, { key: 'prone', name: '倒地' },
    { key: 'restrained', name: '束缚' }, { key: 'stunned', name: '震慑' }, { key: 'unconscious', name: '昏迷' },
  ])
  const [rev, setRev] = React.useState(0)
  const [menu, setMenu] = React.useState(false)
  const [msg, setMsg] = React.useState('')
  const [pDrag, setPDrag] = React.useState(null)   // 面板拖动：{id, slot, index, x, y, over:{slot,index,bar}}
  const [theme, setTheme, presets] = useTheme()
  const rootRef = React.useRef(null)
  const dragRef = React.useRef(null)
  const saveTimer = React.useRef(null)
  const slotRefs = React.useRef({})
  const panelRefs = React.useRef({})
  const pDragRef = React.useRef(null)
  const layoutRef = React.useRef(null)
  React.useEffect(() => { pDragRef.current = pDrag }, [pDrag])
  React.useEffect(() => { layoutRef.current = layout }, [layout])

  React.useEffect(() => {
    call('ws.get', {}).then(r => {
      const l = normLayout(r && r.ok ? r.layout : DEFAULT_LAYOUT)
      setLayout(l)
      setActiveId(l.activeSheet || null)
    }).catch(() => setLayout(normLayout(DEFAULT_LAYOUT)))
  }, [])
  React.useEffect(() => {
    call('party.list', {}).then(r => { if (r && r.ok) setParty(r.items || []) }).catch(() => { })
  }, [rev])
  React.useEffect(() => {
    call('ai.tools', {}).then(r => { if (r && r.ok) setToolCount(r.count) }).catch(() => { })
  }, [])
  // 经验/升级轮询：有人可升级就自动把「升级」面板放出来
  React.useEffect(() => {
    const tick = () => call('party.xp', {}).then(r => {
      if (!r || !r.ok) return
      const m = {}; (r.items || []).forEach(x => { m[x.id] = x })
      setXpMap(m); setXpReady(r.ready || [])
    }).catch(() => { })
    tick()
    const t = (typeof setInterval === 'function') ? setInterval(tick, 5000) : null
    return () => { if (t) clearInterval(t) }
  }, [rev])
  // 有经验满的角色：顶栏给一个「✨ 升级」按钮（点了直接打开他的角色卡）
  React.useEffect(() => {
    if (!xpReady.length) return
    setMsg('✨ ' + xpReady.map(x => x.name + ' 可以升到 Lv' + x.targetLevel).join('、') + ' —— 点右上「✨ 升级」去升级或兼职')
  }, [xpReady.length])

  function commit(next, opts) {
    setLayout(next)
    if (opts && opts.noSave) return
    if (typeof setInterval !== 'function') return
    if (saveTimer.current) clearInterval(saveTimer.current)
    saveTimer.current = setInterval(() => {
      clearInterval(saveTimer.current); saveTimer.current = null
      call('ws.set', { layout: next }).then(r => { if (r && r.ok && r.saved === false) setMsg(r.summary || '布局未落盘') }).catch(() => { })
    }, 400)
  }
  const pick = (id) => { setActiveId(id); const n = cloneLayout(layout); n.activeSheet = id; commit(n) }
  // 用函数声明（会提升）：effect 里会在组件体执行后立刻调用它，const 箭头函数会踩 TDZ
  function openPanel(pid) {
    const n = cloneLayout(layout)
    if (inLayout(n, pid)) { for (const [k] of WS_SLOTS) if (n.slots[k].panels.indexOf(pid) >= 0) n.collapsed = n.collapsed.filter(x => x !== pid); commit(n); return }
    const slot = pid === 'map' ? 'center' : (pid === 'builder' || pid === 'rules' || pid === 'stats' || pid === 'settings' || pid === 'levelup' ? 'right' : 'left')
    n.slots[slot].panels.push(pid)
    n.weights[slot] = n.slots[slot].panels.map(() => 1)
    commit(n)
  }
  const closePanel = (pid) => {
    const n = cloneLayout(layout)
    for (const [k] of WS_SLOTS) {
      const i = n.slots[k].panels.indexOf(pid)
      if (i >= 0) { n.slots[k].panels.splice(i, 1); n.weights[k].splice(i, 1) }
    }
    if (n.maximized === pid) n.maximized = null
    commit(n)
  }
  const movePanel = (pid, slot) => {
    const n = cloneLayout(layout)
    for (const [k] of WS_SLOTS) { const i = n.slots[k].panels.indexOf(pid); if (i >= 0) { n.slots[k].panels.splice(i, 1); n.weights[k].splice(i, 1) } }
    n.slots[slot].panels.push(pid)
    n.weights[slot] = n.slots[slot].panels.map(() => 1)
    if (n.collapsed.indexOf(pid) >= 0) n.collapsed = n.collapsed.filter(x => x !== pid)
    commit(n)
  }
  const addTo = (pid) => {
    const cur = slotOf(layout, pid)
    const order = WS_SLOTS.map(([k]) => k)
    const next = order[(order.indexOf(cur) + 1) % order.length]
    const n = cloneLayout(layout)
    n.slots[next].panels.push(pid)
    n.weights[next] = n.slots[next].panels.map(() => 1)
    commit(n)
    setMsg('已复制到「' + (WS_SLOTS.find(s => s[0] === next) || [])[1] + '」（同一面板可以同时出现在多个槽位）')
  }
  // 拖到指定槽位的指定位置（同槽位内即重排）
  const movePanelTo = (pid, toSlot, toIndex) => {
    const n = reorderPanels(layoutRef.current || layout, pid, toSlot, toIndex)
    commit(n)
    const at = n.slots[toSlot] ? n.slots[toSlot].panels.indexOf(pid) : 0
    setMsg('已移动到「' + ((WS_SLOTS.find(s => s[0] === toSlot) || [])[1] || toSlot) + '」第 ' + (at + 1) + ' 位')
  }
  // 计算拖动落点：命中哪个槽位、插到第几位、指示条画在哪
  const computeDrop = (x, y) => {
    const base = layoutRef.current
    if (!base) return null
    for (const pair of WS_SLOTS) {
      const name = pair[0]
      const el = slotRefs.current[name]
      if (!el) continue
      const r = el.getBoundingClientRect()
      if (x < r.left || x > r.right || y < r.top || y > r.bottom) continue
      const panels = base.slots[name].panels
      const horiz = name === 'bottom'
      for (let i = 0; i < panels.length; i++) {
        const pe = panelRefs.current[name + ':' + i]
        if (!pe) continue
        const pr = pe.getBoundingClientRect()
        const mid = horiz ? (pr.left + pr.right) / 2 : (pr.top + pr.bottom) / 2
        if ((horiz ? x : y) < mid) {
          return { slot: name, index: i, bar: horiz ? { x: pr.left - 2, y: pr.top + 3, w: 4, h: Math.max(8, pr.height - 6) } : { x: pr.left + 6, y: pr.top - 2, w: Math.max(8, pr.width - 12), h: 4 } }
        }
      }
      const lastIdx = panels.length - 1
      const last = lastIdx >= 0 ? panelRefs.current[name + ':' + lastIdx] : null
      if (last) {
        const lr = last.getBoundingClientRect()
        return { slot: name, index: panels.length, bar: horiz ? { x: lr.right - 2, y: lr.top + 3, w: 4, h: Math.max(8, lr.height - 6) } : { x: lr.left + 6, y: lr.bottom - 2, w: Math.max(8, lr.width - 12), h: 4 } }
      }
      return { slot: name, index: 0, bar: horiz ? { x: r.left + 8, y: r.top + 6, w: 4, h: Math.max(8, r.height - 12) } : { x: r.left + 8, y: r.top + 6, w: Math.max(8, r.width - 16), h: 4 } }
    }
    return null
  }
  const startPanelDrag = (pid, slot, index) => (ev) => {
    if (ev && ev.preventDefault) ev.preventDefault()
    setPDrag({ id: pid, slot: slot, index: index, x: ev.clientX, y: ev.clientY, x0: ev.clientX, y0: ev.clientY, moved: false, over: null })
  }
  React.useEffect(() => {
    if (!pDrag || typeof window === 'undefined') return undefined
    const onMove = (ev) => {
      ev.preventDefault()
      setPDrag(d => {
        if (!d) return d
        const moved = d.moved || (Math.abs(ev.clientX - d.x0) + Math.abs(ev.clientY - d.y0) > 5)
        return Object.assign({}, d, { x: ev.clientX, y: ev.clientY, moved: moved, over: moved ? computeDrop(ev.clientX, ev.clientY) : null })
      })
    }
    const onUp = () => {
      const d = pDragRef.current
      // 只有真的拖动过才换位，避免「点一下标题栏就把面板挪走」
      if (d && d.moved && d.over) movePanelTo(d.id, d.over.slot, d.over.index)
      setPDrag(null)
    }
    const onKey = (e) => { if (e.key === 'Escape') setPDrag(null) }
    window.addEventListener('pointermove', onMove, { passive: false })
    window.addEventListener('pointerup', onUp)
    window.addEventListener('keydown', onKey)
    try { document.body.style.userSelect = 'none'; document.body.style.cursor = 'grabbing' } catch (e) { }
    return () => {
      window.removeEventListener('pointermove', onMove)
      window.removeEventListener('pointerup', onUp)
      window.removeEventListener('keydown', onKey)
      try { document.body.style.userSelect = ''; document.body.style.cursor = '' } catch (e) { }
    }
  }, [!!pDrag])
  const toggleCollapse = (pid) => {
    const n = cloneLayout(layout)
    n.collapsed = n.collapsed.indexOf(pid) >= 0 ? n.collapsed.filter(x => x !== pid) : n.collapsed.concat([pid])
    commit(n)
  }
  const toggleMax = (pid) => { const n = cloneLayout(layout); n.maximized = n.maximized === pid ? null : pid; commit(n) }
  const reset = () => { call('ws.reset', {}).then(r => { setLayout(normLayout(r && r.ok ? r.layout : DEFAULT_LAYOUT)); setMsg('布局已重置') }).catch(() => setMsg('重置失败')) }

  // 拖拽分隔条
  const startDrag = (kind, slot, index) => (ev) => {
    ev.preventDefault()
    const root = rootRef.current
    if (!root) return
    const rect = root.getBoundingClientRect()
    dragRef.current = { kind, slot, index, rect, x: ev.clientX, y: ev.clientY, base: cloneLayout(layout) }
    const onMove = (e) => {
      const d = dragRef.current
      if (!d) return
      const n = cloneLayout(d.base)
      const dx = e.clientX - d.x, dy = e.clientY - d.y
      if (d.kind === 'col') {
        if (d.slot === 'left') n.slots.left.size = Math.max(150, Math.min(d.rect.width - 320, d.base.slots.left.size + dx))
        else if (d.slot === 'right') n.slots.right.size = Math.max(200, Math.min(d.rect.width - 320, d.base.slots.right.size - dx))
      } else if (d.kind === 'bottom') {
        n.slots.bottom.size = Math.max(0, Math.min(d.rect.height - 200, d.base.slots.bottom.size - dy))
      } else if (d.kind === 'panel') {
        const slot = d.slot, idx = d.index
        const panels = n.slots[slot].panels
        if (panels.length < 2) return
        const w = (n.weights[slot] || []).slice()
        const delta = (d.slot === 'bottom' ? dx : dy) / Math.max(60, (d.slot === 'bottom' ? d.rect.width : d.rect.height)) * 6
        const a = Math.max(0.4, (d.base.weights[slot][idx] || 1) + delta)
        const b = Math.max(0.4, (d.base.weights[slot][idx + 1] || 1) - delta)
        w[idx] = a; w[idx + 1] = b
        n.weights[slot] = w
      }
      setLayout(n)
    }
    const onUp = () => {
      const cur = dragRef.current
      dragRef.current = null
      window.removeEventListener('pointermove', onMove)
      window.removeEventListener('pointerup', onUp)
      if (cur) {
        setLayout(prev => { if (prev) call('ws.set', { layout: prev }).catch(() => { }); return prev })
      }
    }
    window.addEventListener('pointermove', onMove)
    window.addEventListener('pointerup', onUp)
  }

  if (!layout) return h('div', { className: 'dndp-root ' + themeClassOf(theme), style: themeStyleOf(theme) }, h('div', { className: 'dndp-header' }, h('span', { className: 'dndp-ic' }, '⚔'), h('span', null, '工作区'), h('div', { className: 'dndp-sp' }), h('span', { className: 'dndp-chip' }, '加载布局…')))

  const panelProps = { layout, activeId, rev, onOpen: openPanel }
  const renderPanel = (pid, slot, index, total) => {
    const collapsed = layout.collapsed.indexOf(pid) >= 0
    const body = (() => {
      if (collapsed) return null
      if (pid === 'roster') return h(Roster, { activeId: activeId, onPick: pick, onOpen: openPanel, rev: rev, xpMap: xpMap })
      if (pid === 'sheet') return h(SheetHost, { activeId: activeId, rev: rev, onReload: () => setRev(r => r + 1) })
      if (pid === 'map') return h(MapPanel, { bare: true })
      if (pid === 'log') return h(LogView, { bare: true, party: party, onBack: null })
      if (pid === 'ai') return h(AIPanel, { onOpen: openPanel })
      if (pid === 'combat') return h(CombatPanel, { onChanged: () => setRev(r => r + 1), conditionCatalog: conditionCatalog })
      if (pid === 'dice') return h(DicePanel, { theme: theme })
      if (pid === 'settings') return h(SettingsPanel, { toolCount: toolCount, theme: theme, onTheme: async (t) => { const nt = setTheme(t); const r = await call('theme.set', { theme: nt }); return r }, presets: presets })
      if (pid === 'rules') return h('div', { style: { padding: '10px 12px' } }, h(RulesPanel, {}))
      if (pid === 'stats') return h(StatsPanel, {})
      if (pid === 'builder') return h(Wizard, { onDone: () => { setRev(r => r + 1); closePanel('builder') }, onBack: () => closePanel('builder') })
      return null
    })()
    const weight = (layout.weights[slot] || [])[index] || 1
    const style = collapsed ? { flex: 'none' } : { flex: weight + ' 1 0', minHeight: 0, minWidth: 0 }
    return h('div', {
      key: slot + '-' + pid + '-' + index,
      ref: el => { panelRefs.current[slot + ':' + index] = el },
      className: 'dndp-panelslot',
      style: { display: 'flex', flexDirection: 'column', minHeight: 0, minWidth: 0, ...style },
    },
      h(PanelFrame, {
        id: pid, layout: layout, collapsed: collapsed, isMax: layout.maximized === pid, style: { flex: 1, minHeight: 0 },
        dragging: !!(pDrag && pDrag.id === pid && pDrag.slot === slot),
        onDragStart: startPanelDrag(pid, slot, index),
        onCollapse: () => toggleCollapse(pid), onMax: () => toggleMax(pid), onClose: closePanel, onMove: (s) => movePanel(pid, s), onAddTo: addTo,
      }, body),
      index < total - 1 ? h('div', { className: 'dndp-ws-split ' + (slot === 'bottom' ? 'v' : 'h'), onPointerDown: startDrag('panel', slot, index) }) : null)
  }

  const col = (slotName) => {
    const s = layout.slots[slotName]
    const panels = s.panels
    const horiz = slotName === 'bottom'
    const glowing = !!(pDrag && pDrag.over && pDrag.over.slot === slotName)
    return h('div', { className: 'dndp-ws-inner ' + (horiz ? 'h' : '') + ' dndp-slot' + (glowing ? ' glow' : ''), style: { flex: 1, minHeight: 0, minWidth: 0 } },
      panels.length ? panels.map((pid, i) => renderPanel(pid, slotName, i, panels.length)) : h('div', { className: 'dndp-empty', style: { flex: 1, padding: '10px 6px' } }, '（空）把面板拖到这里'))
  }

  if (layout.maximized && inLayout(layout, layout.maximized)) {
    const pid = layout.maximized
    return h('div', { className: 'dndp-ws ' + themeClassOf(theme), style: themeStyleOf(theme), ref: rootRef },
      h('div', { className: 'dndp-ws-top' }, h('span', { className: 'logo' }, '⚔ SoloTRPG'), h('span', { className: 'dndp-chip' }, PANEL_DEFS[pid].title + ' · 最大化'), h('div', { className: 'dndp-sp' }),
        h('button', { className: 'dndp-btn tiny', onClick: () => toggleMax(pid) }, '还原布局')),
      h('div', { className: 'dndp-ws-main' }, h('div', { style: { flex: 1, display: 'flex', minHeight: 0, padding: 6 } }, renderPanel(pid, slotOf(layout, pid) || 'center', 0, 1))))
  }

  // 空槽位也留出可拖入的区域（左侧窄条 / 右侧窄条 / 拖动时才出现的底栏）
  const hasBottom = layout.slots.bottom.panels.length > 0
  const bottomH = hasBottom ? Math.max(80, layout.slots.bottom.size) : 0
  const leftEmpty = !layout.slots.left.panels.length
  const rightEmpty = !layout.slots.right.panels.length
  const refOf = (name) => (el => { slotRefs.current[name] = el })
  const mainRow = h('div', { className: 'dndp-ws-row', style: { flex: 1, minHeight: 0 } },
    h('div', {
      className: 'dndp-ws-col dndp-slotwrap' + (leftEmpty ? ' slim' : ''),
      ref: refOf('left'),
      style: { width: leftEmpty ? 118 : layout.slots.left.size, flex: 'none', padding: 4 },
    }, col('left')),
    (!leftEmpty || !rightEmpty || layout.slots.center.panels.length) ? h('div', { className: 'dndp-ws-split v', onPointerDown: startDrag('col', 'left') }) : null,
    h('div', { className: 'dndp-ws-col', ref: refOf('center'), style: { flex: 1, minWidth: 0, padding: 4 } }, col('center')),
    h('div', { className: 'dndp-ws-split v', onPointerDown: startDrag('col', 'right') }),
    h('div', {
      className: 'dndp-ws-col dndp-slotwrap' + (rightEmpty ? ' slim' : ''),
      ref: refOf('right'),
      style: { width: rightEmpty ? 132 : layout.slots.right.size, flex: 'none', padding: 4 },
    }, col('right')))

  const allPanels = Object.keys(PANEL_DEFS)
  const menuItems = [
    h('div', { key: 'h', className: 'hd' }, '显示面板'),
    ...allPanels.map(pid => h('button', { key: pid, onClick: () => { openPanel(pid); setMenu(false) } }, (PANEL_DEFS[pid].icon || '') + ' ' + PANEL_DEFS[pid].title + (inLayout(layout, pid) ? '（已在布局中）' : ''))),
    h('div', { key: 's', className: 'sep' }),
    h('button', { key: 'r', onClick: () => { reset(); setMenu(false) } }, '重置为默认布局'),
  ]

  const ghost = pDrag ? h('div', { className: 'dndp-dragghost', style: { left: pDrag.x, top: pDrag.y } },
    ((PANEL_DEFS[pDrag.id] || {}).icon || '▫') + ' ' + ((PANEL_DEFS[pDrag.id] || {}).title || pDrag.id)) : null
  const dropBar = (pDrag && pDrag.over && pDrag.over.bar) ? h('div', { className: 'dndp-drop', style: { position: 'fixed', left: pDrag.over.bar.x, top: pDrag.over.bar.y, width: pDrag.over.bar.w, height: pDrag.over.bar.h } }) : null

  return h('div', { className: 'dndp-ws ' + themeClassOf(theme) + (pDrag ? ' dragging' : ''), style: themeStyleOf(theme), ref: rootRef },
    h('div', { className: 'dndp-ws-top' },
      h('span', { className: 'logo' }, '⚔ SoloTRPG'),
      h('span', { className: 'dndp-chip' }, '工作区'),
      activeId ? h('span', { className: 'dndp-chip' }, '当前角色：' + (activeId || '')) : null,
      h('div', { className: 'dndp-sp' }),
      msg ? h('span', { className: 'dndp-note hl' }, msg) : null,
      xpReady.length ? h('button', {
        className: 'dndp-btn tiny primary', title: '经验已满，去升级或兼职',
        onClick: () => { const x = xpReady[0]; if (x && x.id) { pick(x.id) } ; openPanel('sheet') },
      }, '✨ 升级（' + xpReady[0].name + '）') : null,
      h('button', { className: 'dndp-btn tiny', onClick: () => setRev(r => r + 1) }, '刷新数据'),

const DEFAULT_THEME = { name: 'azure', mode: 'dark', accent: '#5aa0ff', lineBoost: true }
const THEME_LIST = [['azure', '幽蓝'], ['emerald', '翡翠'], ['amber', '琥珀'], ['violet', '紫罗兰'], ['rose', '绯樱'], ['slate', '石板灰']]
const ACCENT_SWATCH = ['#5aa0ff', '#4fbf8b', '#e0a64a', '#a98bff', '#e07a98', '#93a7bd', '#4dd0e1', '#ff8a65']
let THEME_CACHE = null
function normThemeLocal(t) {
  const o = Object.assign({}, DEFAULT_THEME, t || {})
  if (!THEME_LIST.some(x => x[0] === o.name)) o.name = 'azure'
  if (o.mode !== 'light') o.mode = 'dark'
  if (!/^#[0-9a-fA-F]{6}$/.test(String(o.accent || ''))) o.accent = '#5aa0ff'
  return o
}
// 强调色上的文字用黑还是白：按相对亮度自动选
function onAccentOf(hex) {
  const h = String(hex || '').replace('#', '')
  if (h.length !== 6) return '#0d1015'
  const r = parseInt(h.slice(0, 2), 16) / 255, g = parseInt(h.slice(2, 4), 16) / 255, b = parseInt(h.slice(4, 6), 16) / 255
  const lin = (c) => (c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4))
  const L = 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b)
  return L > 0.45 ? '#0d1015' : '#ffffff'
}
function themeClassOf(t) { const th = normThemeLocal(t); return 'dndp-theme-' + th.name + (th.mode === 'light' ? ' dndp-light' : '') }
function themeStyleOf(t) { const th = normThemeLocal(t); return { '--dsw-alias-brand-primary': th.accent, '--dndp-on-accent': onAccentOf(th.accent) } }
function applyThemeGlobally(t) {
  const th = normThemeLocal(t)
  THEME_CACHE = th
  try {
    if (typeof document !== "undefined" && document.documentElement) {
      const de = document.documentElement
      de.style.setProperty('--solo-accent', th.accent)
      de.style.setProperty('--solo-on-accent', onAccentOf(th.accent))
      de.style.setProperty('--solo-mode', th.mode)
    }
    if (typeof window !== 'undefined' && window.dispatchEvent) window.dispatchEvent(new CustomEvent('solotrpg-theme', { detail: th }))
  } catch (e) { }
}
function useTheme() {
  const [th, setTh] = React.useState(THEME_CACHE || normThemeLocal(DEFAULT_THEME))
  const [presets, setPresets] = React.useState(null)
  React.useEffect(() => {
    if (THEME_CACHE) return
    call('theme.get', {}).then(r => {
      if (r && r.ok) { const t = normThemeLocal(r.theme); setTh(t); applyThemeGlobally(t) }
      if (r && r.ok && r.presets) setPresets(r.presets)
    }).catch(() => { })
  }, [])
  const set = (t) => { const nt = normThemeLocal(t); setTh(nt); applyThemeGlobally(nt); return nt }
  return [th, set, presets]
}

// ============ 物理骰子盘 ============
const DIE_SIDES = { d4: 3, d6: 4, d8: 4, d10: 6, d12: 5, d20: 6, d100: 6 }
const DIE_COLOR = { d4: '#b4544a', d6: '#4a7cb4', d8: '#5a9c5f', d10: '#a06ab4', d12: '#c08a3a', d20: '#c2944a', d100: '#6a7a8a' }
const DICE_KINDS = ['d4', 'd6', 'd8', 'd10', 'd12', 'd20', 'd100']
function buildExpr(pool, mod) {
  const parts = DICE_KINDS.filter(k => (pool[k] || 0) > 0).map(k => pool[k] + k)
  const base = parts.join('+') || '1d20'
  const m = Number(mod) || 0
  return base + (m ? (m > 0 ? '+' + m : String(m)) : '')
}
function DicePanel(props) {
  const [pool, setPool] = React.useState({ d20: 1 })
  const [mod, setMod] = React.useState('0')
  const [adv, setAdv] = React.useState('normal')
  const [busy, setBusy] = React.useState(false)
  const [res, setRes] = React.useState(null)
  const [toy, setToy] = React.useState([])
  const [pending, setPending] = React.useState([])
  const [records, setRecords] = React.useState([])
  const [msg, setMsg] = React.useState('')
  const canvasRef = React.useRef(null)
  const bodiesRef = React.useRef([])
  const rafRef = React.useRef(null)
  const lastRef = React.useRef(0)

  const loadPending = () => call('dice.pending', {}).then(r => { if (r && r.ok) setPending(r.pending || []) }).catch(() => { })
  const loadRecords = () => call('dice.results', { limit: 12 }).then(r => { if (r && r.ok) setRecords(r.items || []) }).catch(() => { })
  React.useEffect(() => { loadPending(); loadRecords() }, [])
  React.useEffect(() => {
    const t = (typeof setInterval === 'function') ? setInterval(() => { loadPending(); loadRecords() }, 3000) : null
    return () => { if (t) clearInterval(t) }
  }, [])

  const draw = () => { /* 与之前一致：画骰盘 */ }
  const drawReal = () => {
    const cv = canvasRef.current
    if (!cv || typeof cv.getContext !== 'function') return
    const ctx = cv.getContext('2d')
    if (!ctx) return
    const dpr = (typeof window !== 'undefined' && window.devicePixelRatio) || 1
    const W = cv.clientWidth || 320, H = cv.clientHeight || 150
    if (cv.width !== Math.round(W * dpr) || cv.height !== Math.round(H * dpr)) { cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr) }
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
    ctx.clearRect(0, 0, W, H)
    const light = props.theme && props.theme.mode === 'light'
    ctx.fillStyle = light ? '#e7ecf2' : '#20252c'
    ctx.fillRect(0, 0, W, H)
    ctx.strokeStyle = 'rgba(120,150,190,.35)'
    ctx.lineWidth = 2
    ctx.strokeRect(1, 1, W - 2, H - 2)
    for (const b of bodiesRef.current) {
      ctx.save(); ctx.translate(b.x, b.y); ctx.rotate(b.ang)
      const sides = DIE_SIDES[b.kind] || 6
      ctx.beginPath()
      for (let i = 0; i < sides; i++) { const a = -Math.PI / 2 + i * 2 * Math.PI / sides; const px = Math.cos(a) * b.r, py = Math.sin(a) * b.r; if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py) }
      ctx.closePath()
      ctx.fillStyle = DIE_COLOR[b.kind] || '#666'; ctx.fill()
      ctx.lineWidth = 2; ctx.strokeStyle = 'rgba(255,255,255,.5)'; ctx.stroke()
      ctx.restore()
      ctx.fillStyle = '#fff'
      ctx.font = 'bold ' + Math.max(10, Math.round(b.r * 1.05)) + 'px ui-monospace, Menlo, monospace'
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle'
      ctx.fillText(b.settled && b.value != null ? String(b.value) : '?', b.x, b.y)
    }
  }
  const step = (dt, W, H) => {
    const bodies = bodiesRef.current
    for (const b of bodies) {
      if (b.settled) continue
      b.vy += 1500 * dt
      b.x += b.vx * dt; b.y += b.vy * dt; b.ang += b.va * dt
      b.vx *= 0.994; b.vy *= 0.997; b.va *= 0.986
      if (b.x - b.r < 4) { b.x = 4 + b.r; b.vx = Math.abs(b.vx) * 0.55 }
      if (b.x + b.r > W - 4) { b.x = W - 4 - b.r; b.vx = -Math.abs(b.vx) * 0.55 }
      if (b.y + b.r > H - 4) { b.y = H - 4 - b.r; b.vy = -Math.abs(b.vy) * 0.5; b.vx *= 0.9; b.va *= 0.88 }
      if (b.y - b.r < 4) { b.y = 4 + b.r; b.vy = Math.abs(b.vy) * 0.5 }
    }
    for (let i = 0; i < bodies.length; i++) {
      for (let j = i + 1; j < bodies.length; j++) {
        const a = bodies[i], b = bodies[j]
        const dx = b.x - a.x, dy = b.y - a.y
        const d = Math.sqrt(dx * dx + dy * dy) || 0.01
        const min = a.r + b.r
        if (d < min) {
          const nx = dx / d, ny = dy / d, push = (min - d) / 2
          a.x -= nx * push; a.y -= ny * push; b.x += nx * push; b.y += ny * push
          const p = (a.vx - b.vx) * nx + (a.vy - b.vy) * ny
          if (p > 0) { const imp = p * 0.55; a.vx -= imp * nx; a.vy -= imp * ny; b.vx += imp * nx; b.vy += imp * ny }
        }
      }
    }
    for (const b of bodies) {
      if (b.settled) continue
      if ((b.y + b.r > H - 40) && Math.sqrt(b.vx * b.vx + b.vy * b.vy) < 26 && Math.abs(b.va) < 1.6) { b.settled = true; b.vx = b.vy = b.va = 0 }
    }
    return bodies.every(b => b.settled)
  }
  const animate = (expr, vals, onDone) => {
    const cv = canvasRef.current
    const W = (cv && cv.clientWidth) || 320, H = (cv && cv.clientHeight) || 150
    const list = (vals || []).slice(0, 14)
    const n2 = Math.max(1, list.length)
    const kindGuess = ((expr || '').match(/(\d*)d(\d+)/) || [])[2]
    bodiesRef.current = list.map((v, i) => ({
      kind: 'd' + (kindGuess || 20),
      x: W * (0.18 + 0.64 * (n2 === 1 ? 0.5 : i / (n2 - 1))), y: -30 - i * 8,
      r: n2 > 6 ? 12 : (n2 > 3 ? 15 : 20),
      vx: (Math.random() * 2 - 1) * 460, vy: -180 - Math.random() * 220,
      ang: Math.random() * 6.28, va: (Math.random() * 2 - 1) * 12, value: v, settled: false,
    }))
    const t0 = (typeof performance !== 'undefined' ? performance.now() : Date.now())
    const tick = (now) => {
      const dt = Math.min(0.032, (now - (lastRef.current || now)) / 1000)
      lastRef.current = now
      const all = step(dt, W, H)
      drawReal()
      if (!all && now - t0 < 5000) { rafRef.current = (typeof requestAnimationFrame === 'function') ? requestAnimationFrame(tick) : null; if (!rafRef.current) onDone(); }
      else onDone()
    }
    if (typeof requestAnimationFrame === 'function') { lastRef.current = 0; rafRef.current = requestAnimationFrame(tick) }
    else { bodiesRef.current.forEach(b => { b.settled = true }); onDone() }
  }

  // ① 应 DM 要求投掷（唯一会计入战斗记录的骰子）
  const rollRequested = async (req) => {
    if (busy || !req) return
    setBusy(true); setMsg('')
    const r = await call('roll.dice', { expr: req.expr, advantage: req.advantage || 'normal', append: false })
    if (!r || !r.ok) { setMsg('投掷失败：' + ((r && r.error) || '')); setBusy(false); return }
    animate(req.expr, r.dice, async () => {
      const ans = await call('dice.answer', { id: req.id, dice: r.dice, dropped: r.dropped, total: r.total, detail: r.detail })
      if (ans && ans.ok) {
        setRes({ expr: req.expr, total: r.total, detail: r.detail, requested: true, dc: req.dc, success: ans.success, label: req.label || req.why })
        setMsg(ans.summary)
      } else setMsg('记录失败：' + ((ans && ans.error) || ''))
      setBusy(false); loadPending(); loadRecords()
    })
  }
  // ② 随手丢（玩具，不写记录、不计入统计）
  const rollToy = async () => {
    if (busy) return
    const expr = buildExpr(pool, mod)
    setBusy(true); setMsg('')
    const r = await call('roll.dice', { expr: expr, advantage: adv, append: false })
    if (!r || !r.ok) { setMsg('投掷失败：' + ((r && r.error) || '')); setBusy(false); return }
    animate(expr, r.dice, () => {
      setRes({ expr: expr, total: r.total, detail: r.detail, requested: false })
      setToy(list => [{ expr: expr, total: r.total, detail: r.detail, t: new Date().toLocaleTimeString() }].concat(list).slice(0, 8))
      setBusy(false)
    })
  }
  React.useEffect(() => {
    drawReal()
    const cv = canvasRef.current
    if (!cv || typeof ResizeObserver === 'undefined') return undefined
    const ro = new ResizeObserver(() => drawReal())
    ro.observe(cv)
    return () => { ro.disconnect(); if (rafRef.current && typeof cancelAnimationFrame === 'function') cancelAnimationFrame(rafRef.current) }
  }, [])

  const bump = (k, d) => setPool(p => { const v = Math.max(0, Math.min(12, (p[k] || 0) + d)); const n2 = Object.assign({}, p, { [k]: v }); if (!DICE_KINDS.some(x => n2[x] > 0)) n2.d20 = 1; return n2 })
  const total = DICE_KINDS.reduce((a, k) => a + (pool[k] || 0), 0)
  const req = pending[0] || null

  return h('div', { className: 'dndp-fill' },
    h('div', { className: 'dndp-aibar' },
      h('span', { className: 'dndp-chip' }, req ? ('待投 ' + pending.length + ' 条') : '无待投'),
      h('span', { className: 'dndp-note' }, '战斗判定骰由 DM 发起，玩家点投'),
      h('div', { className: 'dndp-sp' }),
      h('button', { className: 'dndp-btn tiny', onClick: () => { loadPending(); loadRecords() } }, '刷新')),
    h('div', { className: 'dndp-dicebox' },
      // DM 发起的投骰：有请求时按钮亮起
      req ? h('div', { key: 'req', className: 'dndp-req' },
        h('div', null, '📣 DM 要求你投：', h('b', null, req.label || req.why || '（未注明用途）'), req.dc != null ? ('（DC ' + req.dc + '）') : ''),
        h('div', { className: 'dndp-note', style: { marginTop: 2 } }, h('code', null, req.expr + (req.advantage && req.advantage !== 'normal' ? (' ' + req.advantage) : '')) + (req.why && req.label ? ('　' + req.why) : '')),
        h('div', { style: { marginTop: 6 } },
          h('button', { className: 'dndp-dicego live', disabled: busy, onClick: () => rollRequested(req) }, busy ? '翻滚中…' : ('🎲 投掷 ' + req.expr + '（会计入战斗记录）')),
          h('div', { style: { marginTop: 4 } }, h('button', { className: 'dndp-btn tiny', onClick: () => call('dice.clear', { all: true }).then(() => { loadPending(); loadRecords() }) }, '忽略这条要求')))) : null,
      h('div', { key: 'go', style: { marginTop: 8 } },
        h('button', { className: 'dndp-dicego' + (req ? ' live' : ''), disabled: busy || !req, onClick: () => rollRequested(req) }, req ? ('🎲 投掷 ' + req.expr) : '等待 DM 发起投骰要求…')),
      pending.length > 1 ? h('div', { key: 'more', className: 'dndp-note' }, '还有 ' + (pending.length - 1) + ' 条待投（投完当前这条会自动轮到下一条）') : null,
      res ? h('div', { key: 'res', className: 'dndp-note hl', style: { marginTop: 6 } }, (res.requested ? '【战斗判定】' : '') + (res.label ? (res.label + '：') : '') + res.detail + (res.dc != null ? ('｜DC ' + res.dc + ' → ' + (res.success ? '成功' : '失败')) : '')) : null,
      msg ? h('div', { key: 'm', className: 'dndp-note hl' }, msg) : null,
      h('canvas', { key: 'cv', ref: canvasRef, style: { width: '100%', height: 130, borderRadius: 10, display: 'block', marginTop: 8 } }),
      // 记录：最新在最上
      h('div', { key: 'rec', style: { marginTop: 8 } },
        h('div', { className: 'dndp-label' }, '战斗判定记录（最新在上）'),
        records.length ? records.map(r2 => h('div', { key: r2.id, className: 'dndp-rec' },
          h('span', { className: 'tm' }, r2.t ? new Date(r2.t).toLocaleTimeString() : ''),
          h('span', { className: 'lb' }, (r2.actor ? (r2.actor + ' · ') : '') + (r2.label || r2.why || '未注明用途')),
          h('code', { className: 'dndp-logd' }, r2.detail || (r2.expr + ' = ' + r2.total)),
          r2.dc != null ? h('span', { className: r2.success ? 'ok' : 'no' }, 'DC ' + r2.dc + (r2.success ? ' ✔' : ' ✘')) : null)) : h('div', { className: 'dndp-note' }, '还没有战斗判定记录')),
      // 玩具
      h('div', { key: 'toy', style: { marginTop: 10, borderTop: '1px solid var(--dndp-line-soft)', paddingTop: 8 } },
        h('div', { className: 'dndp-label' }, '随手丢（玩具：不写记录、不计入统计）'),
        h('div', { className: 'dndp-chips', style: { marginBottom: 6 } },
          DICE_KINDS.map(k => h('span', { key: k, style: { display: 'inline-flex', alignItems: 'center', gap: 2 } },
            h('button', { className: 'dndp-btn tiny', onClick: () => bump(k, -1), disabled: !(pool[k] > 0) }, '−'),
            h('b', { style: { color: DIE_COLOR[k], minWidth: 22, textAlign: 'center', display: 'inline-block' } }, k + '×' + (pool[k] || 0)),
            h('button', { className: 'dndp-btn tiny', onClick: () => bump(k, 1) }, '+')))),
        h('div', { className: 'dndp-chips' },
          h('span', { className: 'dndp-note' }, '调整值'),
          h('input', { className: 'dndp-input', style: { width: 46 }, value: mod, onChange: e => setMod(e.target.value) }),

          h('button', { className: 'dndp-btn tiny', onClick: () => { setPool({ d20: 1 }); setToy([]) } }, '重置')),
        toy.length ? toy.map((x, i) => h('div', { key: i, className: 'dndp-note' }, x.t + '　' + x.expr + ' = ' + x.total)) : null)))

  if (list && !list.length) rows.push(h('div', { key: 'z', className: 'dndp-empty' }, '暂无角色，点“+ 新建角色”'))
  ;(list || []).forEach((l, i) => rows.push(h('div', { key: i, className: 'dndp-row', onClick: () => open(l.id) }, h('div', { className: 'dndp-ava' }, (l.name || '?').slice(0, 1)), h('div', { className: 'dndp-main' }, h('div', { className: 'dndp-name' }, l.name), h('div', { className: 'dndp-sub' }, (l.race || '') + (l.cls ?
