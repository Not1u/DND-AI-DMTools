          })
        } else {
          const map = await readMap()
          for (const t of (map.tokens || [])) {
            if (t.hidden || t.kind === "object") continue
            const dex = (t.dex == null ? (t.kind === "pc" ? null : 2) : Number(t.dex))
            const init = randInt(20) + (dex == null ? 0 : dex)
            entries.push({ id: "t-" + t.id, name: t.name, kind: t.kind, init: init, hp: t.hp, max: t.max, conditions: [], econ: {}, tokenId: t.id, pcId: null, note: t.note || "", dex: dex })
          }
          try {
            const fs2 = ctx.get("fs"); const dir = await fs2.resolve(CHAR_DIR)
            const names = (typeof fs2.listDir === "function") ? (await fs2.listDir(dir)).filter(e => /[.]json$/i.test(e.name)).map(e => e.name) : []
            const norm = (x) => String(x || "").replace(/[s·.．]/g, "")
            for (const n of names) {
              try {
                const pc = JSON.parse(await fsRead(CHAR_DIR + "/" + n))
                const dex = modOf(Number((pc.abilities || {}).dex) || 10)
                // 如果地图上已经有这个角色的单位（名字相同或互为前缀），合并成一条同一条目
                const hit = entries.find(e => e.kind === "pc" && (e.tokenId === pc.id || norm(e.name) === norm(pc.name) || norm(pc.name).indexOf(norm(e.name)) === 0 || norm(e.name).indexOf(norm(pc.name)) === 0))
                if (hit) {
                  hit.pcId = pc.id
                  hit.name = pc.name
                  hit.dex = dex
                  hit.init = randInt(20) + dex
                  hit.conditions = (hit.conditions || []).concat((pc.conditions || []).map(c => ({ key: c.key, name: c.name, rounds: null })))
                  hit.note = "角色卡+地图单位"
                } else {
                  entries.push({ id: "pc-" + pc.id, name: pc.name, kind: "pc", init: randInt(20) + dex, hp: (pc.hp ? pc.hp.current : null), max: (pc.hp ? pc.hp.max : null), conditions: (pc.conditions || []).map(c => ({ key: c.key, name: c.name, rounds: null })), econ: {}, tokenId: null, pcId: pc.id, note: "", dex: dex })
                }
              } catch (e) { }
            }
          } catch (e) { }
        }
        if (!entries.length) return { ok: false, error: "没有可参战的单位（地图上没有单位，角色卡也读不到）" }
        entries.sort((x, y) => y.init - x.init || String(x.name).localeCompare(String(y.name)))
        const st = { active: true, round: 1, turnIndex: 0, order: entries, notes: a.note || "" }
        const wr = await writeCombat(st)
        await pushLog([{ kind: "turn", actor: "DM", text: "先攻已定：R1 开始", detail: entries.map(e => e.name + " " + e.init).join("｜") }], { campaign: a.campaign, round: 1 })
        return Object.assign(combatView(st), { started: entries.length, saved: !!wr.ok, saveError: wr.error || "", summary: "战斗开始：R1，先攻顺序 " + entries.map(e => e.name + "(" + e.init + ")").join(" → ") })
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("combat.next", async (args) => {
      try {
        const a = args || {}
        const st = await readCombat()
        if (!st.active) return { ok: false, error: "还没开始战斗（combat.start）" }
        const events = []
        const steps = Math.max(1, Math.min(50, Number(a.steps) || 1))
        for (let k = 0; k < steps; k++) {
          let idx = (st.turnIndex + 1) % st.order.length
          if (idx === 0) st.round = (st.round || 1) + 1
          // 换人时：先结算新行动者的状态时间
          const cur = st.order[idx]
          const kept = []
          for (const c of (cur.conditions || [])) {
            if (c.rounds == null) { kept.push(c); continue }
            const left = Number(c.rounds) - 1
            if (left > 0) { kept.push(Object.assign({}, c, { rounds: left })) } else events.push(cur.name + " 的「" + c.name + "」结束了")
          }
          cur.conditions = kept
          cur.econ = {}
          st.turnIndex = idx
          events.push("轮到 " + cur.name + "（R" + st.round + "）")
          if (cur.pcId) await syncConditionsToSheet(cur)
          if (!a.steps) break
        }
        const wr = await writeCombat(st)
        const cur = st.order[st.turnIndex]
        await pushLog([{ kind: "turn", actor: "DM", text: (a.log || ("轮到 " + cur.name)), detail: events.join("；") }], {})
        return Object.assign(combatView(st), { events: events, saved: !!wr.ok, saveError: wr.error || "", summary: events.join("；") })
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("combat.economy", async (args) => {
      try {
        const a = args || {}
        const st = await readCombat()
        let idx = findEntryIndex(st, a.id)
        if (idx < 0 && a.id == null) idx = st.turnIndex
        if (idx < 0) return { ok: false, error: "找不到该战斗单位：" + a.id }
        const e = st.order[idx]
        e.econ = e.econ || {}
        const key = String(a.key || a.kind || "")
        if (!ECON_KEYS.includes(key)) return { ok: false, error: "key 需要 action|bonus|reaction|move" }
        if (a.reset) e.econ = {}
        else if (key === "move") e.econ.move = Math.max(0, (Number(e.econ.move) || 0) + (Number(a.feet) || 0))
        else e.econ[key] = (a.used === false ? false : true)
        const wr = await writeCombat(st)
        const used = ECON_KEYS.filter(k => k !== "move" && e.econ[k]).map(k => ({ action: "动作", bonus: "附赠动作", reaction: "反应" })[k])
        return jclean(Object.assign(combatView(st), { saved: !!wr.ok, summary: e.name + "：" + (used.length ? ("已用 " + used.join("、")) : "行动经济未使用") + (e.econ.move ? ("｜已移动 " + e.econ.move + " 尺") : "") }))
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("combat.conditions", async (args) => {
      try {
        const a = args || {}
        const st = await readCombat()
        const idx = findEntryIndex(st, a.id)
        if (idx < 0) return { ok: false, error: "找不到该战斗单位：" + a.id }
        const e = st.order[idx]
        e.conditions = e.conditions || []
        for (const r of (a.remove || [])) e.conditions = e.conditions.filter(c => c.key !== r)
        for (const c of (a.add || [])) {
          const spec = typeof c === "string" ? { key: c } : c
          const known = CONDITIONS.find(x => x.key === spec.key)
          const item = { key: spec.key, name: spec.name || (known && known.name) || spec.key, level: (spec.level == null ? null : Number(spec.level)),
            rounds: (spec.rounds == null || spec.rounds === "" ? null : Number(spec.rounds)), note: spec.note || "" }
          const at = e.conditions.findIndex(c2 => c2.key === item.key)
          if (at >= 0) e.conditions[at] = item; else e.conditions.push(item)
        }
        const wr = await writeCombat(st)
        if (e.pcId) await syncConditionsToSheet(e)
        await pushLog([{ kind: "state", actor: e.name, text: a.log || "状态变化", detail: e.conditions.map(c => c.name + (c.level != null ? " " + c.level : "") + (c.rounds ? "（剩 " + c.rounds + " 回合）" : "")).join("、") || "（无状态）" }], {})
        return jclean(Object.assign(combatView(st), { saved: !!wr.ok, summary: e.name + " 状态：" + (e.conditions.map(c => c.name + (c.rounds ? "(" + c.rounds + ")" : "")).join("、") || "无") }))
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("combat.damage", async (args) => {
      try {
        const a = args || {}
        const st = await readCombat()
        const idx = findEntryIndex(st, a.id)
        if (idx < 0) return { ok: false, error: "找不到该战斗单位：" + a.id }
        const e = st.order[idx]
        const amount = Number(a.amount) || 0
        if (e.hp != null) {
          e.hp = Math.max(0, Math.min(e.max == null ? 9999 : e.max, Number(e.hp) + amount))
        }
        // 同步到地图单位 + 角色卡
        if (e.tokenId) { try { const map = await readMap(); const tk = (map.tokens || []).find(t => t.id === e.tokenId); if (tk && tk.hp != null) { tk.hp = Math.max(0, Math.min(tk.max || 9999, Number(tk.hp) + amount)); const wr2 = await writeMap(map); void wr2 } } catch (e2) { } }
        if (e.pcId) { try { const pc = JSON.parse(await fsRead(CHAR_DIR + "/" + String(e.pcId).replace(/[^A-Za-z0-9._-]/g, "") + ".json")); pc.hp.current = Math.max(0, Math.min(pc.hp.max, Number(pc.hp.current) + amount)); pc.meta = pc.meta || {}; pc.meta.updatedAt = nowIso(); await persist(pc) } catch (e2) { } }
        const wr = await writeCombat(st)
        await pushLog([{ kind: amount < 0 ? "damage" : "heal", actor: e.name, text: a.log || (amount < 0 ? "受到伤害" : "恢复生命"), detail: (amount > 0 ? "+" : "") + amount + " → HP " + e.hp + (e.max ? "/" + e.max : "") }], {})
        return jclean(Object.assign(combatView(st), { saved: !!wr.ok, summary: e.name + " HP " + e.hp + (e.max ? "/" + e.max : "") + "（" + (amount > 0 ? "+" : "") + amount + "）" }))
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("combat.end", async () => {
      try {
        const st = await readCombat()
        st.active = false; st.turnIndex = 0; st.round = 0; st.order = (st.order || []).map(e => Object.assign({}, e, { econ: {}, conditions: [] }))
        const wr = await writeCombat(st)
        return jclean(Object.assign(combatView(st), { saved: !!wr.ok, summary: "战斗结束" }))
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    // ===== 骰子请求：AI 弹提示，玩家点一下就投；结果回写战斗记录供 AI 读取 =====
    const readDice = async () => {
      try { const j = JSON.parse(await fsRead(DICE_PATH)); if (j && Array.isArray(j.pending)) return j } catch (e) { }
      return { seq: 0, pending: [], done: [] }
    }
    const writeDice = async (d) => { d.updatedAt = nowIso(); d.done = (d.done || []).slice(-60); d.pending = (d.pending || []).slice(-20); return await writeRaw(DICE_PATH, JSON.stringify(d, null, 2)) }
    unreg.push(harness.handle("dice.request", async (args) => {
      try {
        const a = args || {}
        if (!a.expr) return { ok: false, error: "需要 expr，例如 1d20+2" }
        if (!a.label && !a.why) return { ok: false, error: "必须写清这次投骰的用途：label（例如「敏捷豁免（火球术）」）" }
        const d = await readDice()
        d.seq = (Number(d.seq) || 0) + 1
        const req = {
          id: "r" + d.seq, t: nowIso(), expr: String(a.expr), advantage: (a.advantage === "adv" || a.advantage === "dis") ? a.advantage : "normal",
          label: a.label || "", actor: a.actor || "", dc: (a.dc == null || a.dc === "" ? null : Number(a.dc)), kind: a.kind || "roll", why: a.why || "",
        }
        d.pending.push(req)
        const wr = await writeDice(d)
        return jclean({ ok: true, id: req.id, request: req, saved: !!wr.ok, saveError: wr.error || "", summary: "已请求玩家投掷：" + req.expr + (req.dc ? ("（DC " + req.dc + "）") : "") + (req.label ? (" — " + req.label) : "") })
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("dice.pending", async () => {
      try { const d = await readDice(); return jclean({ ok: true, pending: d.pending || [], done: (d.done || []).slice(-10), count: (d.pending || []).length }) }
      catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("dice.answer", async (args) => {
      try {
        const a = args || {}
        const d = await readDice()
        const req = (d.pending || []).find(r => r.id === a.id)
        if (!req) return { ok: false, error: "没有这条待投请求：" + a.id }
        d.pending = d.pending.filter(r => r.id !== a.id)
        const entry = { id: req.id, t: nowIso(), expr: req.expr, advantage: req.advantage, label: req.label, why: req.why || "", actor: req.actor || a.actor || "",
          dc: req.dc, dice: a.dice || [], dropped: a.dropped || [], total: (a.total == null ? null : Number(a.total)), detail: a.detail || "",
          success: (req.dc == null || a.total == null) ? null : (Number(a.total) >= Number(req.dc)), casual: false }
        d.done.push(entry)
        const wr = await writeDice(d)
        await pushLog([{ kind: req.kind || "roll", actor: entry.actor || "玩家", text: (req.label || "") + (req.dc ? ("（DC " + req.dc + "）") : "") + (entry.success === true ? " → 成功" : entry.success === false ? " → 失败" : ""), detail: entry.detail }], {})
        return jclean({ ok: true, id: req.id, total: entry.total, success: entry.success, detail: entry.detail, dc: req.dc, saved: !!wr.ok, summary: "已记录：" + entry.detail + (req.dc != null ? ("｜DC " + req.dc + " → " + (entry.success ? "成功" : "失败")) : "") })
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }

        // ↑ reverse：最新的一条排最前，方便一眼看到刚投的是什么
        return jclean({ ok: true, count: items.length, items: items, pending: (d.pending || []).length, summary: items.length ? items.map(x => (x.actor || "玩家") + " " + x.expr + " = " + x.total + (x.dc != null ? (" vs DC " + x.dc + (x.success ? " 成功" : " 失败")) : "")).join("；") : "还没有玩家投掷记录" })
      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    unreg.push(harness.handle("dice.clear", async (args) => {
      try { const a = args || {}; const d = await readDice(); if (a.all) { d.pending = []; d.done = [] } else { d.done = [] } const wr = await writeDice(d); return { ok: true, saved: !!wr.ok, summary: a.all ? "已清空请求与记录" : "已清空记录" } }
      catch (e) { return { ok: false, error: String(e && e.message || e) } }

    // ===== 主题（界面配色）：存 data/theme.json =====    // ===== 主题（界面配色）：存 data/theme.json =====    // ===== 主题（界面配色）：存 data/theme.json =====    // ===== 主题（界面配色）：存 data/theme.json =====
    const THEME_PATH = ROOT + '/data/theme.json'

      if (!THEME_PRESETS[o.name]) o.name = 'azure'
      if (o.mode !== 'light') o.mode = 'dark'
      if (!/^#[0-9a-fA-F]{6}$/.test(String(o.accent || ''))) o.accent = THEME_PRESETS[o.name].accent

      } catch (e) { return { ok: false, error: String(e && e.message || e) } }
    }))
    return () => { unreg.forEach(f => f()) }
  }
